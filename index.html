วิธีเชื่อมต่อ Google Sheets

ในการบันทึกข้อมูลลง Google Sheets ให้ทำตามขั้นตอนต่อไปนี้:

  1. ไปที่ Google Apps Script แล้วกดสร้าง "โครงการใหม่ (New Project)"
  2. ตั้งชื่อโปรเจกต์ตามต้องการ เช่น "API บันทึกซองงานแต่ง"
  3. คัดลอกโค้ดด้านล่างนี้ ไปวางทับโค้ดเดิมทั้งหมดในหน้า Script:
function doPost(e) {
  try {
    // กำหนดรหัส Sheet ID ของคุณ และชื่อ Sheet ที่ต้องการบันทึก
    const sheetId = '1hPg0KA4MeoYqUiewRCLimQH7f7pFh7ijGf_m3BLsYWA';
    const sheetName = 'Data';
    
    // เปิดไฟล์ Google Sheet ตาม ID ที่ระบุ
    const ss = SpreadsheetApp.openById(sheetId);
    let sheet = ss.getSheetByName(sheetName);
    
    // ถ้าย้งไม่มี Sheet ชื่อ Data ให้สร้างใหม่และใส่หัวตาราง
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(['Timestamp', 'ชื่อ-นามสกุล', 'จำนวนเงิน', 'ประเภท', 'ฝ่าย', 'หมายเหตุ']);
      // ไม่ต้อง return ออก ให้ทำงานรับข้อมูลต่อเลย
    }

    // รับข้อมูลที่ส่งมาจากฟอร์มเว็บ
    const timestamp = new Date();
    const name = e.parameter.name || '';
    const amount = e.parameter.amount || 0;
    const type = e.parameter.type || '';
    const side = e.parameter.side || '';
    const note = e.parameter.note || '';

    // เพิ่มข้อมูลลงแถวใหม่
    sheet.appendRow([timestamp, name, amount, type, side, note]);

    // ตอบกลับหน้าเว็บว่าสำเร็จ
    return ContentService.createTextOutput(JSON.stringify({result: 'success'}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({result: 'error', error: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. คลิกปุ่ม ทำให้ใช้งานได้ (Deploy) มุมขวาบน > การทำให้ใช้งานได้รายการใหม่ (New deployment)
